```python
def solve(numbers: list, target: int) -> List[int]:
    """"""
    hash_dict: dict = {}
    
    for i, num in enumerate(numbers):
        res = target - num
        if res in hash_dict:
            return [hash_dict[res], i]
        else:
            hash_dict[num] = i
    
    return []
```
